# E-commerce project — Python dependencies
# Install with:  pip install -r requirements.txt
#
# Versions below are what this project was built and tested against.
# "X>=a,b" means "a or newer, but below b".

Django>=6.1,<6.2          # web framework (LTS line 5.2 works too)
Pillow>=12.0              # image handling for product photo uploads
django-crispy-forms>=2.3  # renders Django forms with a CSS framework
crispy-bootstrap5>=2025.1 # Bootstrap 5 template pack for crispy-forms
python-dotenv>=1.0        # reads SECRET_KEY, DB config etc. from a .env file

# PHASE 20 — production server & database:
# (both install fine on Windows too; gunicorn simply isn't USED there,
#  psycopg2-binary is only loaded when the database is PostgreSQL)
gunicorn>=23.0,<24           # production WSGI server (Linux/macOS)
# 2.9.10 is the floor on purpose: Neon routes connections with TLS SNI, and
# the libpq bundled in psycopg2 only supports SNI from v14 onwards. Older
# builds fail to connect with the baffling "Endpoint ID is not specified".
psycopg2-binary>=2.9.10      # PostgreSQL driver (Neon / production database)
whitenoise>=6.6              # serves /static/ in production (no Nginx needed)
